-- Ativa a extensão necessária para gerar UUIDs
CREATE EXTENSION IF NOT EXISTS pgcrypto;

-- ==========================
-- Tabela de Países
-- ==========================
CREATE TABLE countries (
    id_country UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    country_name VARCHAR(100) NOT NULL UNIQUE,
    continent VARCHAR(100) NOT NULL
);

-- ==========================
-- Tabela de Setores
-- ==========================
CREATE TABLE sectors (
    id_sector UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    sector_name VARCHAR(100) NOT NULL UNIQUE
);

-- ==========================
-- Tabela de Empresas
-- ==========================
CREATE TABLE companies (
    id_company UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    company_name VARCHAR(150) NOT NULL,
    id_country UUID NOT NULL,
    id_sector UUID NOT NULL,

    CONSTRAINT fk_company_country
        FOREIGN KEY (id_country)
        REFERENCES countries(id_country),

    CONSTRAINT fk_company_sector
        FOREIGN KEY (id_sector)
        REFERENCES sectors(id_sector)
);

-- ==========================
-- Tabela de Registros de Emissão
-- ==========================
CREATE TABLE emission_records (
    id_record UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    id_company UUID NOT NULL,
    emission_year INT NOT NULL,
    co2_emission_kt NUMERIC(12,2) NOT NULL,

    CONSTRAINT fk_emission_company
        FOREIGN KEY (id_company)
        REFERENCES companies(id_company)
);

DQL

-- =========================================
-- 1. Mapeamento Corporativo
-- Nome da empresa, setor e país
-- =========================================

SELECT
    c.company_name,
    s.sector_name,
    co.country_name
FROM companies c
INNER JOIN sectors s
    ON c.id_sector = s.id_sector
INNER JOIN countries co
    ON c.id_country = co.id_country;

-- =========================================
-- 2. Empresas do setor Energia
-- Ano e emissão, do mais recente ao mais antigo
-- =========================================

SELECT
    c.company_name,
    er.emission_year,
    er.co2_emission_kt
FROM companies c
INNER JOIN sectors s
    ON c.id_sector = s.id_sector
INNER JOIN emission_records er
    ON c.id_company = er.id_company
WHERE s.sector_name = 'Energia'
ORDER BY er.emission_year DESC;

-- =========================================
-- 3. País que mais emitiu CO2
-- Soma total das emissões
-- =========================================

SELECT
    co.country_name,
    SUM(er.co2_emission_kt) AS total_emissions
FROM countries co
INNER JOIN companies c
    ON co.id_country = c.id_country
INNER JOIN emission_records er
    ON c.id_company = er.id_company
GROUP BY co.country_name
ORDER BY total_emissions DESC;
-- =========================================
-- 4. Top 3 maiores emissores em 2024
-- =========================================

SELECT
    co.country_name,
    SUM(er.co2_emission_kt) AS total_emissions
FROM countries co
INNER JOIN companies c
    ON co.id_country = c.id_country
INNER JOIN emission_records er
    ON c.id_company = er.id_company
GROUP BY co.country_name
ORDER BY total_emissions DESC
LIMIT 3;

-- ======================================

-- 5. Média de emissão por setor
SELECT
    s.sector_name,
    AVG(er.co2_emission_kt) AS media_emissao
FROM sectors s
INNER JOIN companies c
ON s.id_sector = c.id_sector
INNER JOIN emission_records er
ON c.id_company = er.id_company
GROUP BY s.sector_name
ORDER BY media_emissao DESC;

--6. Evolução histórica das emissões
SELECT
    emission_year,
    SUM(co2_emission_kt) AS total_global
FROM emission_records
GROUP BY emission_year
ORDER BY emission_year;
